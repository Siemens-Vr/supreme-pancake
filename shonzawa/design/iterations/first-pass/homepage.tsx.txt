function QuoteButton({
  service,
  children = "Request a Quote",
  className = "",
  light = false,
}: {
  service?: string;
  children?: React.ReactNode;
  className?: string;
  light?: boolean;
}) {
  return (
    <a
      href={quoteUrl(service)}
      target="_blank"
      rel="noopener noreferrer"
      className={`button ${light ? "button-white" : "button-primary"} ${className}`}
      aria-label={`Request a quote${service ? ` for ${service}` : ""} on WhatsApp (opens in a new tab)`}
    >
      {children}
      <ArrowUpRight size={17} aria-hidden="true" />
    </a>
  );
}
